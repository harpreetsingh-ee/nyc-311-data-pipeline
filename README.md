# nyc-311-data-pipeline
data engineering learnings

copy data from GCP project (ee-india-se-data) to the target warehouse (Snowflake).

GCS bucket: harpreet_singh_nyc311


## Snowflake queries

SHOW STAGES;

SHOW INTEGRATIONS
DESCRIBE INTEGRATION gcs_nyc311_integration;

LIST @DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW."HARPREET_SINGH_nyc_311_stage"

SHOW GRANTS ON INTEGRATION gcs_nyc311_integration;

SELECT CURRENT_ORGANIZATION_NAME() || '-' || CURRENT_ACCOUNT_NAME();


SELECT * from DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW.NYC_311_DATASET LIMIT 10
DESCRIBE TABLE DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW.NYC_311_DATASET

DESCRIBE FILE FORMAT DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW."HARPREET_SINGH_nyc_311_csv_format"

DESCRIBE PIPE DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW."HARPREET_SINGH_nyc_311_pipe"

SELECT SYSTEM$PIPE_STATUS('DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW."HARPREET_SINGH_nyc_311_pipe"');

TRUNCATE TABLE DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW.NYC_311_DATASET

ALTER PIPE DE_CROSS_SKILLING_NYC_311.HARPREET_SINGH_RAW."HARPREET_SINGH_nyc_311_pipe" REFRESH;
